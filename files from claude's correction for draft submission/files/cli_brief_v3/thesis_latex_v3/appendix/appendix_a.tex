\chapter*{Appendix A: Repository Structure and Reproducibility}

\section*{A.1 Datasets}
\paragraph{Line B benchmark (pre-registered dataset).}
\begin{itemize}
    \item \textbf{File:} \texttt{data/publishable\_scar\_production/multimodal\_publishable.csv}
    \item \textbf{SHA-256:} \texttt{2035777f957fa4c5cc830989f4fe40fb9190589f17c567cb29a3ef0e6b2acae2}.
    \item \textbf{Physiology:} WESAD chest ECG and EDA from 15 participants (S2--S17, excluding S1 and S12), 30-second windows with a 15-second stride, prepared by \texttt{src/prepare\_wesad.py}.
    \item \textbf{Faces:} 418 FFHQ images accepted by an InsightFace screen, with synthetic brow-line scars, masks, and clean counterfactuals, built by \texttt{src/build\_publishable\_scar\_dataset.py}.
    \item \textbf{Split (by participant):} training S2, S3, S4, S7, S8, S9, S13, S14, S16, S17 (2,344 samples); validation S6, S15 (504); test S5, S10, S11 (496).
    \item \textbf{Training association:} $\rho = 0.85$ between scar and stress.
\end{itemize}
\paragraph{Line A cohort.}
\begin{itemize}
    \item UBFC-Phys participants s1--s4, three tasks each (12 clips); T1 (rest) $\rightarrow Y = 0$; T2 and T3 (speech and arithmetic) $\rightarrow Y = 1$.
    \item Outputs: \texttt{outputs/leakage\_run/}.
\end{itemize}

\section*{A.4 Training Configuration}
\begin{itemize}
    \item Seeds: 42, 100, 2026, 777, and 888; 250 epochs per run; batch size 32; learning rate $10^{-4}$.
    \item Backbone: MobileNetV3-Small for all models.
    \item Hardware: RTX~4060 GPU for training; x86-64 CPU for the latency benchmark.
\end{itemize}

\section*{A.5 Orthogonality Verification}
The orthogonality error $\|W_{\text{causal}} W_{\text{conf}}^{\top}\|_F$ can be computed for any checkpoint with \texttt{model.stiefel\_decomp.verify\_mutual\_orthogonality()}. Its measured value at initialization was $1.7149 \times 10^{-6}$.
